### Teste 3.1: Inicialização do Microcontrolador e Comunicação Serial

**Objetivo:** Validar a gravação de firmware básico no microcontrolador ESP32-S3 e verificar a integridade da comunicação Serial via interface USB Native a $115200\text{ baud}$.

#### Código Utilizado:
```cpp
#include <Arduino.h>

void setup() {
  Serial.begin(115200);
  delay(1000);

  Serial.println("\n=============================================");
  Serial.println("  TESTE 3.1: INICIALIZAÇÃO E SERIAL (ESP32-S3) ");
  Serial.println("=============================================");
  Serial.printf("Modelo do Chip: %s\n", ESP.getChipModel());
  Serial.printf("Revisão do Chip: %d\n", ESP.getChipRevision());
  Serial.printf("Frequência CPU: %d MHz\n", ESP.getCpuFreqMHz());
  Serial.println("Status: Comunicação Serial estabelecida com sucesso!");
}

void loop() {
  // Mantém em standby
}
```

#### Resultado Obtido:
- A ESP32-S3 completou a sequência de *boot* limpa, sem reinicializações (*brownout* ou *watchdog reset*), transmitindo com precisão as informações de chip e frequência no Monitor Serial em $115200\text{ baud}$.

### Teste 3.5: Leitura e Calibração dos Sensores Ultrassônicos (HC-SR04)

**Objetivo:** Medir a distância exata até um obstáculo físico e calibrar a leitura com régua graduada.

#### Código Utilizado:
```cpp
#include <Arduino.h>

#define TRIG_ESQ 4
#define ECHO_ESQ 5

const unsigned long TIMEOUT_US = 25000;

float lerDistanciaCM(int pinTrig, int pinEcho) {
  digitalWrite(pinTrig, LOW);
  delayMicroseconds(2);
  digitalWrite(pinTrig, HIGH);
  delayMicroseconds(10);
  digitalWrite(pinTrig, LOW);

  unsigned long duracao = pulseIn(pinEcho, HIGH, TIMEOUT_US);
  if (duracao == 0) return 400.0;
  return (duracao * 0.0343) / 2.0;
}

void setup() {
  Serial.begin(115200);

  pinMode(TRIG_ESQ, OUTPUT);
  pinMode(ECHO_ESQ, INPUT);
  digitalWrite(TRIG_ESQ, LOW);

  Serial.println("\n=============================================");
  Serial.println("  TESTE 3.5: LEITURA DE 1 SENSOR (ESQUERDO)  ");
  Serial.println("=============================================");
}

void loop() {
  float dEsq = lerDistanciaCM(TRIG_ESQ, ECHO_ESQ);
  Serial.printf("Distância Sensor Esquerdo: %.1f cm\n", dEsq);
  delay(2000);
}
```

### Testes 3.6 e 3.7: Odometria e Interrupções dos Encoders (Motores N20)

**Objetivo:** Decodificar os sinais de quadratura dos encoders magnéticos utilizando interrupções alocadas na `IRAM_ATTR`.

#### Código Utilizado:
```cpp
#include <Arduino.h>

#define ENC_ESQ_A 12
#define ENC_ESQ_B 13
#define ENC_DIR_A 14
#define ENC_DIR_B 15

volatile long ticksEsq = 0;
volatile long ticksDir = 0;

void IRAM_ATTR isrEncoderEsq() {
  if (digitalRead(ENC_ESQ_B) == HIGH) ticksEsq++;
  else ticksEsq--;
}

void IRAM_ATTR isrEncoderDir() {
  if (digitalRead(ENC_DIR_B) == HIGH) ticksDir++;
  else ticksDir--;
}

void setup() {
  Serial.begin(115200);

  pinMode(ENC_ESQ_A, INPUT_PULLUP);
  pinMode(ENC_ESQ_B, INPUT_PULLUP);
  pinMode(ENC_DIR_A, INPUT_PULLUP);
  pinMode(ENC_DIR_B, INPUT_PULLUP);

  attachInterrupt(digitalPinToInterrupt(ENC_ESQ_A), isrEncoderEsq, RISING);
  attachInterrupt(digitalPinToInterrupt(ENC_DIR_A), isrEncoderDir, RISING);

  Serial.println("\n=============================================");
  Serial.println("  TESTE 3.6 & 3.7: ODOMETRIA E ENCODERS N20  ");
  Serial.println("=============================================");
}

void loop() {
  Serial.printf("Ticks Esquerda: %ld | Ticks Direita: %ld\n", ticksEsq, ticksDir);
  delay(2000);
}
```

### Teste 3.8: Processamento Digital e Limiarização (*Thresholding*)

**Objetivo:** Validar a tomada de decisão booleana (`SIM`/`NÃO`) para a presença de paredes utilizando um limiar de $5{,}0\text{ cm}$.

#### Código Utilizado:
```cpp
#include <Arduino.h>

#define TRIG_ESQ 4
#define ECHO_ESQ 5

const float LIMITE_PAREDE_CM = 5.0;
const unsigned long TIMEOUT_US = 25000;

float lerDistanciaCM(int pinTrig, int pinEcho) {
  digitalWrite(pinTrig, LOW);
  delayMicroseconds(2);
  digitalWrite(pinTrig, HIGH);
  delayMicroseconds(10);
  digitalWrite(pinTrig, LOW);

  unsigned long duracao = pulseIn(pinEcho, HIGH, TIMEOUT_US);
  if (duracao == 0) return 400.0;
  return (duracao * 0.0343) / 2.0;
}

void setup() {
  Serial.begin(115200);

  pinMode(TRIG_ESQ, OUTPUT);
  pinMode(ECHO_ESQ, INPUT);
  digitalWrite(TRIG_ESQ, LOW);

  Serial.println("\n=============================================");
  Serial.println("  TESTE 3.8: LIMIARIZAÇÃO (1 SENSOR @ 5 CM)  ");
  Serial.println("=============================================");
}

void loop() {
  float dEsq = lerDistanciaCM(TRIG_ESQ, ECHO_ESQ);
  bool paredeEsq = (dEsq < LIMITE_PAREDE_CM);

  Serial.printf("Distância: %.1f cm | PAREDE (< 5 cm): %s\n",
                dEsq, paredeEsq ? "SIM" : "NÃO");
```

### Testes 3.9 e 3.10: Algoritmo Flood Fill e Consumo de Memória SRAM

**Objetivo:** Medir o tempo de execução e a estabilidade da memória RAM no cálculo da navegação do labirinto ($12 \times 4$).

#### Código Utilizado:
```cpp
#include <Arduino.h>

#define MAZE_COLS 4
#define MAZE_ROWS 12
#define INF 9999

uint8_t walls[MAZE_ROWS][MAZE_COLS];
int floodMap[MAZE_ROWS][MAZE_COLS];

struct Cell { int x; int y; };

void floodFill() {
  for (int y = 0; y < MAZE_ROWS; y++) {
    for (int x = 0; x < MAZE_COLS; x++) {
      floodMap[y][x] = INF;
    }
  }

  Cell queue[MAZE_ROWS * MAZE_COLS];
  int front = 0, rear = 0;

  floodMap[11][3] = 0; 
  queue[rear++] = { 3, 11 };

  while (front < rear) {
    Cell current = queue[front++];
    int x = current.x;
    int y = current.y;

    int dx[4] = { 0, 1, 0, -1 };
    int dy[4] = { -1, 0, 1, 0 };

    for (int dir = 0; dir < 4; dir++) {
      int nx = x + dx[dir];
      int ny = y + dy[dir];

      if (nx >= 0 && nx < MAZE_COLS && ny >= 0 && ny < MAZE_ROWS) {
        if (floodMap[ny][nx] == INF) {
          floodMap[ny][nx] = floodMap[y][x] + 1;
          queue[rear++] = { nx, ny };
        }
      }
    }
  }
}

void setup() {
  Serial.begin(115200);
  delay(1000);

  Serial.println("\n=============================================");
  Serial.println("  TESTE 3.9 & 3.10: FLOOD FILL & MEMÓRIA RAM  ");
  Serial.println("=============================================");

  uint32_t ramAntes = ESP.getFreeHeap();
  unsigned long tInicio = micros();

  floodFill();

  unsigned long tExecucao = micros() - tInicio;
  uint32_t ramDepois = ESP.getFreeHeap();

  Serial.printf("Tempo de Execução do Flood Fill: %lu us (microsegundos)\n", tExecucao);
  Serial.printf("Tamanho da Matriz de Paredes: %u bytes\n", sizeof(walls));
  Serial.printf("Tamanho do Mapa Flood Fill: %u bytes\n", sizeof(floodMap));
  Serial.printf("SRAM Livre Antes: %u bytes | SRAM Livre Depois: %u bytes\n", ramAntes, ramDepois);
}

void loop() {
}
```

### Testes 3.11 e 3.12: Acionamento Reversível e Controle PWM (ZS-H2)

**Objetivo:** Confirmar o controle de sentido de rotação e regulação de velocidade dos motores N20 através da ponte H ZS-H2.

#### Código Utilizado:
```cpp
#include <Arduino.h>

#define INA1 16   // Motor Esquerdo 1
#define INA2 17   // Motor Esquerdo 2
#define INB1 18   // Motor Direito 1
#define INB2 21   // Motor Direito 2

void moverMotorEsquerdo(int velocidade) {
  velocidade = constrain(velocidade, -255, 255);
  if (velocidade > 0) {
    analogWrite(INA1, velocidade); analogWrite(INA2, 0);
  } else if (velocidade < 0) {
    analogWrite(INA1, 0); analogWrite(INA2, -velocidade);
  } else {
    analogWrite(INA1, 0); analogWrite(INA2, 0);
  }
}

void moverMotorDireito(int velocidade) {
  velocidade = constrain(velocidade, -255, 255);
  if (velocidade > 0) {
    analogWrite(INB1, velocidade); analogWrite(INB2, 0);
  } else if (velocidade < 0) {
    analogWrite(INB1, 0); analogWrite(INB2, -velocidade);
  } else {
    analogWrite(INB1, 0); analogWrite(INB2, 0);
  }
}

void setup() {
  Serial.begin(115200);

  pinMode(INA1, OUTPUT); pinMode(INA2, OUTPUT);
  pinMode(INB1, OUTPUT); pinMode(INB2, OUTPUT);

  Serial.println("\n=============================================");
  Serial.println("  TESTE 3.11 & 3.12: MOTORES & PWM (ZS-H2)   ");
  Serial.println("=============================================");
}

void loop() {
  Serial.println("1. Girando para FRENTE (PWM 150)...");
  moverMotorEsquerdo(150); moverMotorDireito(150);
  delay(2000);

  Serial.println("2. PARADO...");
  moverMotorEsquerdo(0); moverMotorDireito(0);
  delay(1500);

  Serial.println("3. Girando para TRÁS (PWM -150)...");
  moverMotorEsquerdo(-150); moverMotorDireito(-150);
  delay(2000);

  Serial.println("4. PARADO...");
  moverMotorEsquerdo(0); moverMotorDireito(0);
  delay(2000);
}
```

### Teste 3.13: Validação dos Rádios de Comunicação Sem Fio (Wi-Fi e Bluetooth BLE)

**Objetivo:** Verificar a operação conjunta das interfaces de RF (escaneamento Wi-Fi e transmissão de beacon BLE 5.0) da ESP32-S3.

#### Código Utilizado:
```cpp
#include <Arduino.h>
#include <WiFi.h>
#include <BLEDevice.h>
#include <BLEUtils.h>
#include <BLEServer.h>

#define SERVICE_UUID "4fafc201-1fb5-459e-8fcc-c5c9c331914b"

void setup() {
  Serial.begin(115200);
  delay(1000);

  Serial.println("\n=============================================");
  Serial.println("   TESTE DE HARDWARE: WI-FI E BLE (ESP32-S3) ");
  Serial.println("=============================================");

  // --- 1. TESTE DE WI-FI ---
  Serial.println("\n[1] Iniciando teste do rádio Wi-Fi...");
  WiFi.mode(WIFI_STA);
  WiFi.disconnect();
  delay(100);

  int n = WiFi.scanNetworks();
  if (n == 0) {
    Serial.println("-> Wi-Fi ativo, mas nenhuma rede foi encontrada nas proximidades.");
  } else {
    Serial.printf("-> Wi-Fi OK! %d redes encontradas:\n", n);
    for (int i = 0; i < n && i < 5; ++i) {
      Serial.printf("   %d: %s (%d dBm)\n", i + 1, WiFi.SSID(i).c_str(), WiFi.RSSI(i));
    }
  }

  // --- 2. TESTE DE BLUETOOTH (BLE) ---
  Serial.println("\n[2] Iniciando rádio Bluetooth BLE...");
  BLEDevice::init("MicroPrea_BLE_Test");
  BLEServer *pServer = BLEDevice::createServer();
  BLEService *pService = pServer->createService(SERVICE_UUID);
  pService->start();

  BLEAdvertising *pAdvertising = BLEDevice::getAdvertising();
  pAdvertising->addServiceUUID(SERVICE_UUID);
  pAdvertising->setScanResponse(true);
  BLEDevice::startAdvertising();

  Serial.println("-> BLE OK! Anunciando sinal como: 'MicroPrea_BLE_Test'");
  Serial.println("=============================================\n");
}

void loop() {
  Serial.println("Status: Módulos Wi-Fi e BLE ativos e operando sem falhas.");
  delay(3000);
}
```
